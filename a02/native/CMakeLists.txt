cmake_minimum_required(VERSION 3.22.1)
project(jarvis_llm_a02 LANGUAGES C CXX)
# Same engine configuration as app/src/main/cpp/CMakeLists.txt (protected, unchanged).
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
    message(FATAL_ERROR "Run tools/prepare-native.sh before building")
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" "llama-build" EXCLUDE_FROM_ALL)
add_library(jarvis_llm_a02 SHARED a02_bridge.cpp)
# Sealed-descriptor open shim (see a02_bridge.cpp): only this library, all platforms.
target_link_options(jarvis_llm_a02 PRIVATE "-Wl,--wrap=fopen")
target_include_directories(jarvis_llm_a02 PRIVATE "${LLAMA_SOURCE_DIR}/src")
if(ANDROID)
    target_link_libraries(jarvis_llm_a02 PRIVATE llama android log)
    target_link_options(jarvis_llm_a02 PRIVATE "-Wl,-z,max-page-size=16384")
else()
    find_package(JNI REQUIRED)
    target_compile_definitions(jarvis_llm_a02 PRIVATE JARVIS_HOST_TESTS)
    target_include_directories(jarvis_llm_a02 PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(jarvis_llm_a02 PRIVATE llama)
endif()
# Adversarial snapshot suite: host ctest, and an Android executable run through adb on emulators.
enable_testing()
find_package(Threads REQUIRED)
add_executable(a02_snapshot_test ../tests/a02_snapshot_test.cpp)
target_compile_definitions(a02_snapshot_test PRIVATE _GNU_SOURCE)
target_link_libraries(a02_snapshot_test PRIVATE Threads::Threads)
add_test(NAME a02_snapshot COMMAND a02_snapshot_test)
