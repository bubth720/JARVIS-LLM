cmake_minimum_required(VERSION 3.22.1)
project(jarvis_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    # Optimize CPU kernels even in an installable debug APK; retain native symbols.
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
    message(FATAL_ERROR "Run tools/prepare-native.sh before building")
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" "llama-build" EXCLUDE_FROM_ALL)
if(ANDROID)
    add_library(jarvis_llm SHARED bridge.cpp)
    target_link_libraries(jarvis_llm PRIVATE llama android log)
    target_include_directories(jarvis_llm PRIVATE "${LLAMA_SOURCE_DIR}/src")
    # NDK r28 + explicit ELF alignment; APK zip alignment is checked separately.
    target_link_options(jarvis_llm PRIVATE "-Wl,-z,max-page-size=16384")
else()
    enable_testing()
    find_package(JNI REQUIRED)
    add_library(jarvis_llm SHARED bridge.cpp)
    target_compile_definitions(jarvis_llm PRIVATE JARVIS_HOST_TESTS)
    target_include_directories(jarvis_llm PRIVATE ${JNI_INCLUDE_DIRS} "${LLAMA_SOURCE_DIR}/src")
    target_link_libraries(jarvis_llm PRIVATE llama)

    add_executable(model_memory_probe tests/model_memory_probe.cpp)
    target_include_directories(model_memory_probe PRIVATE "${LLAMA_SOURCE_DIR}/src")
    target_link_libraries(model_memory_probe PRIVATE llama)
    add_executable(native_safety_test tests/native_safety_test.cpp)
    target_link_libraries(native_safety_test PRIVATE llama ggml ggml-cpu)
    add_test(NAME native_safety COMMAND native_safety_test)
endif()
